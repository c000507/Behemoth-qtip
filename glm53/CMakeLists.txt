cmake_minimum_required(VERSION 3.25)
project(glm53 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PCRE2 REQUIRED IMPORTED_TARGET libpcre2-8)
find_package(OpenMP)
option(GLM53_HIP "Build native wave32 ROCm inference kernels" OFF)
option(GLM53_CUDA "Build native CUDA inference kernels (Turing or newer)" OFF)
option(GLM53_SERVER "Build the standalone text HTTP server" OFF)
add_library(glm53_core src/config.cpp src/tokenizer.cpp src/weights.cpp src/model.cpp
  ../runtime/src/checkpoint.cpp ../src/safetensors.cpp)
target_include_directories(glm53_core PUBLIC include ../runtime/include ../third_party ../src)
target_link_libraries(glm53_core PUBLIC PkgConfig::PCRE2)
target_compile_options(glm53_core PRIVATE $<$<NOT:$<COMPILE_LANGUAGE:CUDA>>:-Wall>
  $<$<NOT:$<COMPILE_LANGUAGE:CUDA>>:-Wextra>)
if(GLM53_HIP AND GLM53_CUDA)
 message(FATAL_ERROR "Select one GPU provider per build: GLM53_HIP or GLM53_CUDA")
endif()
if(GLM53_HIP)
 set(CMAKE_HIP_ARCHITECTURES gfx1201 CACHE STRING "GPU architecture")
 enable_language(HIP)
 set(CMAKE_HIP_STANDARD 20)
 find_package(hip REQUIRED CONFIG)
 set_source_files_properties(src/device_ops.cu PROPERTIES LANGUAGE HIP)
 target_sources(glm53_core PRIVATE src/gpu.cpp src/device_ops.cu)
 target_compile_options(glm53_core PRIVATE $<$<COMPILE_LANGUAGE:HIP>:-mno-wavefrontsize64>)
 target_link_libraries(glm53_core PUBLIC hip::host)
 target_compile_definitions(glm53_core PUBLIC GLM53_HIP=1 __HIP_PLATFORM_AMD__=1)
endif()
if(GLM53_CUDA)
 set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "NVIDIA architectures (75 = RTX 2080 Ti)")
 enable_language(CUDA)
 set(CMAKE_CUDA_STANDARD 20)
 set(CMAKE_CUDA_STANDARD_REQUIRED ON)
 find_package(CUDAToolkit REQUIRED)
 target_sources(glm53_core PRIVATE src/gpu.cpp src/device_ops.cu)
 target_link_libraries(glm53_core PUBLIC CUDA::cudart)
 target_compile_definitions(glm53_core PUBLIC GLM53_CUDA=1)
 # NCCL all-reduce for tensor parallelism when the toolkit's NCCL is present (the copy-engine collective
 # remains the fallback and the choice for ranks sharing one GPU).
 option(GLM53_NCCL "Use NCCL for tensor-parallel collectives when available" ON)
 if(GLM53_NCCL)
  find_path(NCCL_INCLUDE_DIR nccl.h HINTS "${NCCL_ROOT}/include")
  find_library(NCCL_LIBRARY NAMES nccl libnccl.so.2 HINTS "${NCCL_ROOT}/lib")
  if(NCCL_INCLUDE_DIR AND NCCL_LIBRARY)
   target_include_directories(glm53_core PRIVATE ${NCCL_INCLUDE_DIR})
   target_compile_definitions(glm53_core PRIVATE GLM53_NCCL=1)
   target_link_libraries(glm53_core PUBLIC ${NCCL_LIBRARY})
   message(STATUS "GLM53 tensor parallelism: NCCL at ${NCCL_LIBRARY}")
  else()
   message(STATUS "GLM53 tensor parallelism: NCCL not found, copy-engine collectives only")
  endif()
 endif()
endif()
if(OpenMP_CXX_FOUND)
  target_link_libraries(glm53_core PUBLIC OpenMP::OpenMP_CXX)
endif()
add_executable(glm53 src/main.cpp)
target_link_libraries(glm53 PRIVATE glm53_core)
add_executable(glm53_tests tests/test.cpp)
target_link_libraries(glm53_tests PRIVATE glm53_core)
target_compile_definitions(glm53_tests PRIVATE GLM53_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
if(GLM53_SERVER)
 find_package(Boost REQUIRED)
 find_package(Threads REQUIRED)
 add_executable(glm53_server src/server.cpp ../runtime/src/reasoning.cpp ../runtime/src/sampling.cpp)
 target_include_directories(glm53_server PRIVATE ${Boost_INCLUDE_DIRS})
 target_compile_definitions(glm53_server PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
 target_link_libraries(glm53_server PRIVATE glm53_core Threads::Threads)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND)
   add_test(NAME glm53_http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py --build ${CMAKE_CURRENT_BINARY_DIR})
 endif()
endif()
enable_testing()
add_test(NAME glm53_correctness COMMAND glm53_tests)
set_tests_properties(glm53_correctness PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=4")
if(GLM53_HIP OR GLM53_CUDA)
 add_executable(glm53_gpu_tests tests/gpu_test.cpp)
 target_link_libraries(glm53_gpu_tests PRIVATE glm53_core)
 add_test(NAME glm53_gpu COMMAND glm53_gpu_tests)
 set_tests_properties(glm53_gpu PROPERTIES SKIP_RETURN_CODE 77)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND)
   add_test(NAME glm53_gpu_models COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/gpu_integration.py --build ${CMAKE_CURRENT_BINARY_DIR})
   set_tests_properties(glm53_gpu_models PROPERTIES SKIP_RETURN_CODE 77)
 endif()
endif()
