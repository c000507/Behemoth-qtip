cmake_minimum_required(VERSION 3.25)
project(qtip_finetune LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(QTIP_TRAIN_HIP "Build native gfx1201 training" OFF)
option(QTIP_TRAIN_TOKENIZER "Build native GM-CoT preparation" ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
find_package(OpenMP COMPONENTS CXX)
add_library(qtip_train_core src/io.cpp src/cpu.cpp src/layer.cpp src/trainer.cpp
  ${REPO}/runtime/src/qtip.cpp ${REPO}/runtime/src/checkpoint.cpp
  ${REPO}/runtime/src/attention.cpp ${REPO}/runtime/src/sorted_layout.cpp
  ${REPO}/src/safetensors.cpp)
target_include_directories(qtip_train_core PUBLIC include ${REPO}/runtime/include
  ${REPO}/third_party PRIVATE ${REPO}/src)
target_link_libraries(qtip_train_core PUBLIC OpenSSL::Crypto Threads::Threads)
target_compile_options(qtip_train_core PRIVATE -Wall -Wextra)
if(OpenMP_CXX_FOUND)
  target_link_libraries(qtip_train_core PRIVATE OpenMP::OpenMP_CXX)
endif()
if(QTIP_TRAIN_HIP)
  set(CMAKE_HIP_ARCHITECTURES gfx1201 CACHE STRING "GPU architecture")
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 20)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  find_package(hip REQUIRED CONFIG)
  target_compile_definitions(qtip_train_core PUBLIC __HIP_PLATFORM_AMD__=1)
  find_package(hipblaslt REQUIRED CONFIG)
  find_package(rocwmma QUIET CONFIG)
  target_sources(qtip_train_core PRIVATE src/hip.cpp src/ops.hip src/flash.hip
    ${REPO}/runtime/src/qtip.hip src/train_kernels.hip)
  set_target_properties(qtip_train_core PROPERTIES HIP_STANDARD 20 HIP_STANDARD_REQUIRED ON)
  if(TARGET roc::rocwmma)
    get_target_property(WMMA_HEADERS roc::rocwmma INTERFACE_INCLUDE_DIRECTORIES)
  else()
    find_path(WMMA_HEADERS rocwmma/rocwmma.hpp REQUIRED
      HINTS ${CMAKE_HIP_COMPILER_ROCM_ROOT}/include /opt/rocm/include)
  endif()
  target_include_directories(qtip_train_core SYSTEM PRIVATE ${WMMA_HEADERS})
  target_link_libraries(qtip_train_core PUBLIC hip::host roc::hipblaslt)
  target_compile_definitions(qtip_train_core PUBLIC QTIP_TRAIN_HIP=1)
  target_compile_options(qtip_train_core PRIVATE $<$<COMPILE_LANGUAGE:HIP>:-mno-wavefrontsize64>)
endif()
add_executable(qtip_finetune src/main.cpp)
target_link_libraries(qtip_finetune PRIVATE qtip_train_core)
add_executable(qtip_check src/check.cpp)
target_link_libraries(qtip_check PRIVATE qtip_train_core)
add_executable(qtip_bench src/bench.cpp)
target_link_libraries(qtip_bench PRIVATE qtip_train_core)
add_executable(qtip_attention_bench src/attention_bench.cpp)
target_link_libraries(qtip_attention_bench PRIVATE qtip_train_core)
add_executable(qtip_train_tests tests/test_training.cpp)
target_link_libraries(qtip_train_tests PRIVATE qtip_train_core)
target_include_directories(qtip_train_tests PRIVATE ${REPO}/src)
enable_testing()
if(QTIP_TRAIN_HIP)
  add_executable(qtip_gpu_tests tests/test_gpu.cpp)
  target_link_libraries(qtip_gpu_tests PRIVATE qtip_train_core)
  add_test(NAME native_gpu COMMAND qtip_gpu_tests)
  set_tests_properties(native_gpu PROPERTIES SKIP_RETURN_CODE 77)
  add_test(NAME native_gpu_tuned COMMAND qtip_gpu_tests)
  set_tests_properties(native_gpu_tuned PROPERTIES SKIP_RETURN_CODE 77 ENVIRONMENT "QTIP_TRAIN_GEMM_TUNE=1")
endif()
add_test(NAME native_training COMMAND qtip_train_tests)
if(QTIP_TRAIN_TOKENIZER)
  find_package(sentencepiece CONFIG QUIET)
  if(NOT TARGET sentencepiece-static AND NOT TARGET sentencepiece)
    include(FetchContent)
    set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
    set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
    set(SPM_ENABLE_TEST OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(spm URL https://github.com/google/sentencepiece/archive/refs/tags/v0.2.1.tar.gz
      URL_HASH SHA256=c1a59e9259c9653ad0ade653dadff074cd31f0a6ff2a11316f67bee4189a8f1b
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(spm)
  endif()
  add_executable(qtip_prepare src/prepare.cpp)
  target_link_libraries(qtip_prepare PRIVATE qtip_train_core)
  if(TARGET sentencepiece-static)
    target_link_libraries(qtip_prepare PRIVATE sentencepiece-static)
  else()
    target_link_libraries(qtip_prepare PRIVATE sentencepiece)
  endif()
  if(spm_SOURCE_DIR)
    target_include_directories(qtip_prepare PRIVATE ${spm_SOURCE_DIR}/src)
  endif()
  if(TARGET sentencepiece_train-static)
    add_executable(qtip_prepare_tests tests/test_prepare.cpp)
    target_link_libraries(qtip_prepare_tests PRIVATE qtip_train_core sentencepiece-static sentencepiece_train-static)
    if(spm_SOURCE_DIR)
      target_include_directories(qtip_prepare_tests PRIVATE ${spm_SOURCE_DIR}/src)
    endif()
    add_test(NAME native_preparation COMMAND qtip_prepare_tests)
  endif()
endif()
