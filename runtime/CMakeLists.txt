cmake_minimum_required(VERSION 3.25)
project(behemoth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BEHEMOTH_ROCM "Build native gfx1201 FP8 kernels" OFF)
option(BEHEMOTH_CUDA "Build CUDA Llama/Behemoth QTIP and GLM-5.3 inference" OFF)
option(BEHEMOTH_CUDA_FAST "Enable NCCL tensor parallelism, cuBLAS and cuDNN prefill" OFF)
option(BEHEMOTH_SERVER "Build HTTP server (requires ROCm)" OFF)
option(BEHEMOTH_TOKENIZER "Build C++ tokenizer and tokenizer tests" OFF)
option(BEHEMOTH_HTTP_TOOLS "Build HTTP load benchmark (no GPU required)" OFF)
option(BEHEMOTH_SANITIZE "Enable host address/undefined sanitizers" OFF)
include(FetchContent)
if(BEHEMOTH_CUDA)
  # GLM and Behemoth exchange JSON across the common CUDA API. Use the same
  # vendored header throughout this build, including behemoth_core.
  add_library(behemoth_cuda_json INTERFACE)
  target_include_directories(behemoth_cuda_json INTERFACE ../third_party)
  add_library(nlohmann_json::nlohmann_json ALIAS behemoth_cuda_json)
else()
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(json)
endif()
find_package(Threads REQUIRED)
add_library(behemoth_core src/attention.cpp src/cache.cpp src/checkpoint.cpp src/sorted_layout.cpp src/scheduler.cpp src/sampling.cpp src/reasoning.cpp src/protocol.cpp src/reference_block.cpp src/qtip.cpp src/prefix_cache.cpp src/gbnf.cpp src/json_schema.cpp src/constraint.cpp)
target_include_directories(behemoth_core PUBLIC include)
target_link_libraries(behemoth_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(behemoth_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
find_package(OpenMP COMPONENTS CXX)
if(OpenMP_CXX_FOUND)
  target_link_libraries(behemoth_core PRIVATE OpenMP::OpenMP_CXX)
endif()
if(BEHEMOTH_SANITIZE)
  target_compile_options(behemoth_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(behemoth_core PUBLIC -fsanitize=address,undefined)

if(BEHEMOTH_ROCM)
  set(CMAKE_HIP_ARCHITECTURES gfx1201 CACHE STRING "AMD GPU architecture")
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 20)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  find_package(hip REQUIRED CONFIG)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  find_package(rocwmma REQUIRED CONFIG)
  find_package(hipblaslt REQUIRED CONFIG)
  find_package(rccl REQUIRED CONFIG)
  add_library(behemoth_gpu src/attention.hip src/projection.hip src/ops.hip src/transport.hip src/kv_transfer.hip src/qtip.hip src/qtip_dense.hip src/fp8_linear.hip src/model.cpp src/engine.cpp)
  # rocWMMA is header-only. Its packaged target exports unconditional '-x hip'
  # flags, which must not leak into host C++ sources or downstream executables.
  get_target_property(ROCWMMA_HEADERS roc::rocwmma INTERFACE_INCLUDE_DIRECTORIES)
  target_include_directories(behemoth_gpu SYSTEM PRIVATE ${ROCWMMA_HEADERS})
  target_link_libraries(behemoth_gpu PUBLIC behemoth_core hip::host roc::hipblaslt roc::rccl)
  target_compile_options(behemoth_gpu PRIVATE $<$<COMPILE_LANGUAGE:HIP>:-mno-wavefrontsize64>)
  target_compile_definitions(behemoth_gpu PUBLIC BEHEMOTH_ROCM=1)
  add_executable(behemoth_gpu_tests tests/gpu_tests.cpp)
  target_link_libraries(behemoth_gpu_tests PRIVATE behemoth_gpu)
  add_test(NAME gpu_correctness COMMAND behemoth_gpu_tests)
  set_tests_properties(gpu_correctness PROPERTIES LABELS "gpu")
  foreach(variant IN ITEMS native legacy prefill_v1)
    add_test(NAME attention_audit_${variant} COMMAND behemoth_gpu_tests --attention-audit)
    set_tests_properties(attention_audit_${variant} PROPERTIES LABELS "gpu;attention")
  endforeach()
  set_tests_properties(attention_audit_native PROPERTIES ENVIRONMENT "BEHEMOTH_ATTN_LEGACY=0;BEHEMOTH_ATTN_PREFILL_V1=0")
  set_tests_properties(attention_audit_legacy PROPERTIES ENVIRONMENT "BEHEMOTH_ATTN_LEGACY=1;BEHEMOTH_ATTN_PREFILL_V1=0")
  set_tests_properties(attention_audit_prefill_v1 PROPERTIES ENVIRONMENT "BEHEMOTH_ATTN_LEGACY=0;BEHEMOTH_ATTN_PREFILL_V1=1")
  add_test(NAME gpu_transport_correctness COMMAND behemoth_gpu_tests --two-gpu)
  set_tests_properties(gpu_transport_correctness PROPERTIES LABELS "gpu;multi_gpu")
  add_executable(behemoth_bench tools/gpu_bench.cpp)
  target_link_libraries(behemoth_bench PRIVATE behemoth_gpu)
  add_executable(behemoth_projection_bench tools/projection_bench.cpp)
  target_link_libraries(behemoth_projection_bench PRIVATE behemoth_gpu)
  add_executable(behemoth_model_tests tests/model_tests.cpp)
  target_link_libraries(behemoth_model_tests PRIVATE behemoth_gpu)
  add_executable(behemoth_block_tests tests/block_tests.cpp)
  target_link_libraries(behemoth_block_tests PRIVATE behemoth_gpu)
  add_executable(behemoth_qtip_bench tools/qtip_bench.cpp)
  target_link_libraries(behemoth_qtip_bench PRIVATE behemoth_gpu)
  add_executable(behemoth_prefill_bench tools/prefill_bench.cpp)
  target_link_libraries(behemoth_prefill_bench PRIVATE behemoth_gpu)
  add_executable(behemoth_qtip_tests tests/qtip_tests.cpp)
  target_link_libraries(behemoth_qtip_tests PRIVATE behemoth_gpu OpenMP::OpenMP_CXX)
  add_test(NAME qtip_correctness COMMAND behemoth_qtip_tests)
  set_tests_properties(qtip_correctness PROPERTIES LABELS "gpu")
endif()
