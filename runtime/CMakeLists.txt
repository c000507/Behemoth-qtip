cmake_minimum_required(VERSION 3.25)
project(behemoth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BEHEMOTH_ROCM "Build native gfx1201 FP8 kernels" OFF)
option(BEHEMOTH_SERVER "Build HTTP server (requires ROCm)" OFF)
option(BEHEMOTH_TOKENIZER "Build C++ tokenizer and tokenizer tests" OFF)
option(BEHEMOTH_HTTP_TOOLS "Build HTTP load benchmark (no GPU required)" OFF)
option(BEHEMOTH_SANITIZE "Enable host address/undefined sanitizers" OFF)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
find_package(Threads REQUIRED)
add_library(behemoth_core src/attention.cpp src/cache.cpp src/checkpoint.cpp src/sorted_layout.cpp src/scheduler.cpp src/sampling.cpp src/protocol.cpp src/reference_block.cpp src/qtip.cpp src/prefix_cache.cpp)
target_include_directories(behemoth_core PUBLIC include)
target_link_libraries(behemoth_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(behemoth_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
find_package(OpenMP COMPONENTS CXX)
if(OpenMP_CXX_FOUND)
  target_link_libraries(behemoth_core PRIVATE OpenMP::OpenMP_CXX)
endif()
if(BEHEMOTH_SANITIZE)
  target_compile_options(behemoth_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(behemoth_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(behemoth_prefill_compute tools/prefill_compute.cpp)
target_link_libraries(behemoth_prefill_compute PRIVATE nlohmann_json::nlohmann_json)
add_executable(behemoth_inspect tools/inspect.cpp)
target_link_libraries(behemoth_inspect PRIVATE behemoth_core)
add_executable(behemoth_reference_bench tools/reference_bench.cpp)
target_link_libraries(behemoth_reference_bench PRIVATE behemoth_core)
enable_testing()
add_executable(behemoth_tests tests/core_tests.cpp)
target_link_libraries(behemoth_tests PRIVATE behemoth_core)
add_test(NAME core_correctness COMMAND behemoth_tests)
if(BEHEMOTH_ROCM)
  set(CMAKE_HIP_ARCHITECTURES gfx1201 CACHE STRING "AMD GPU architecture")
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 20)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  find_package(hip REQUIRED CONFIG)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  find_package(rocwmma REQUIRED CONFIG)
  find_package(hipblaslt REQUIRED CONFIG)
  find_package(rccl REQUIRED CONFIG)
  add_library(behemoth_gpu src/attention.hip src/projection.hip src/ops.hip src/transport.hip src/kv_transfer.hip src/qtip.hip src/qtip_dense.hip src/model.cpp src/engine.cpp)
  # rocWMMA is header-only. Its packaged target exports unconditional '-x hip'
  # flags, which must not leak into host C++ sources or downstream executables.
  get_target_property(ROCWMMA_HEADERS roc::rocwmma INTERFACE_INCLUDE_DIRECTORIES)
  target_include_directories(behemoth_gpu SYSTEM PRIVATE ${ROCWMMA_HEADERS})
  target_link_libraries(behemoth_gpu PUBLIC behemoth_core hip::host roc::hipblaslt roc::rccl)
  target_compile_options(behemoth_gpu PRIVATE $<$<COMPILE_LANGUAGE:HIP>:-mno-wavefrontsize64>)
  target_compile_definitions(behemoth_gpu PUBLIC BEHEMOTH_ROCM=1)
  add_executable(behemoth_gpu_tests tests/gpu_tests.cpp)
  target_link_libraries(behemoth_gpu_tests PRIVATE behemoth_gpu)
  add_test(NAME gpu_correctness COMMAND behemoth_gpu_tests)
  set_tests_properties(gpu_correctness PROPERTIES LABELS "gpu")
  add_test(NAME gpu_transport_correctness COMMAND behemoth_gpu_tests --two-gpu)
  set_tests_properties(gpu_transport_correctness PROPERTIES LABELS "gpu;multi_gpu")
  add_executable(behemoth_bench tools/gpu_bench.cpp)
  target_link_libraries(behemoth_bench PRIVATE behemoth_gpu)
  add_executable(behemoth_projection_bench tools/projection_bench.cpp)
  target_link_libraries(behemoth_projection_bench PRIVATE behemoth_gpu)
  add_executable(behemoth_model_tests tests/model_tests.cpp)
  target_link_libraries(behemoth_model_tests PRIVATE behemoth_gpu)
  add_executable(behemoth_block_tests tests/block_tests.cpp)
  target_link_libraries(behemoth_block_tests PRIVATE behemoth_gpu)
  add_executable(behemoth_qtip_bench tools/qtip_bench.cpp)
  target_link_libraries(behemoth_qtip_bench PRIVATE behemoth_gpu)
  add_executable(behemoth_prefill_bench tools/prefill_bench.cpp)
  target_link_libraries(behemoth_prefill_bench PRIVATE behemoth_gpu)
  add_executable(behemoth_qtip_tests tests/qtip_tests.cpp)
  target_link_libraries(behemoth_qtip_tests PRIVATE behemoth_gpu OpenMP::OpenMP_CXX)
  add_test(NAME qtip_correctness COMMAND behemoth_qtip_tests)
  set_tests_properties(qtip_correctness PROPERTIES LABELS "gpu")
endif()
if(BEHEMOTH_SERVER OR BEHEMOTH_TOKENIZER)
  set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
  set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(sentencepiece URL https://github.com/google/sentencepiece/archive/refs/tags/v0.2.1.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(sentencepiece)
  add_library(behemoth_tokenizer src/tokenizer.cpp)
  target_include_directories(behemoth_tokenizer PRIVATE ${sentencepiece_SOURCE_DIR}/src)
  target_link_libraries(behemoth_tokenizer PUBLIC behemoth_core PRIVATE sentencepiece-static)
  add_executable(behemoth_tokenizer_tests tests/tokenizer_tests.cpp)
  target_link_libraries(behemoth_tokenizer_tests PRIVATE behemoth_tokenizer)
endif()
if(BEHEMOTH_SERVER)
  if(NOT BEHEMOTH_ROCM)
    message(FATAL_ERROR "The server requires BEHEMOTH_ROCM; no CPU/mock generation fallback exists")
  endif()
  find_path(Boost_INCLUDE_DIR boost/beast.hpp REQUIRED)
  add_executable(behemoth_server src/server.cpp)
  target_include_directories(behemoth_server PRIVATE ${Boost_INCLUDE_DIR})
  target_link_libraries(behemoth_server PRIVATE behemoth_gpu behemoth_tokenizer)
  target_compile_definitions(behemoth_server PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
  add_executable(behemoth_eval tools/eval.cpp)
  target_link_libraries(behemoth_eval PRIVATE behemoth_gpu behemoth_tokenizer)
endif()
if(BEHEMOTH_SERVER OR BEHEMOTH_HTTP_TOOLS)
  find_path(Boost_INCLUDE_DIR boost/beast.hpp REQUIRED)
  add_executable(behemoth_load_bench tools/load_bench.cpp)
  target_include_directories(behemoth_load_bench PRIVATE ${Boost_INCLUDE_DIR})
  target_link_libraries(behemoth_load_bench PRIVATE behemoth_core)
  target_compile_definitions(behemoth_load_bench PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
  add_executable(behemoth_http_tests tests/http_tests.cpp)
  target_include_directories(behemoth_http_tests PRIVATE ${Boost_INCLUDE_DIR})
  target_link_libraries(behemoth_http_tests PRIVATE behemoth_core)
  target_compile_definitions(behemoth_http_tests PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
endif()
