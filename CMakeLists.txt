cmake_minimum_required(VERSION 3.25)
option(QTIP_CUDA "Build the quantizer GPU pipeline for NVIDIA CUDA (sm_86, cuBLASLt) instead of ROCm HIP" OFF)
if(QTIP_CUDA)
  # distro nvcc 12.0 supports gcc <= 12 as the host compiler; pick one if the user did not choose
  if(NOT DEFINED CMAKE_CUDA_HOST_COMPILER AND NOT DEFINED ENV{CUDAHOSTCXX} AND NOT DEFINED ENV{NVCC_CCBIN})
    find_program(QTIP_CUDA_HOST_GXX NAMES g++-12 g++-11)
    if(QTIP_CUDA_HOST_GXX)
      set(CMAKE_CUDA_HOST_COMPILER ${QTIP_CUDA_HOST_GXX})
    endif()
  endif()
  set(CMAKE_CUDA_ARCHITECTURES 86 CACHE STRING "NVIDIA GPU architecture (RTX A4000 = 86)")
  project(qtip_quant LANGUAGES CXX CUDA)
else()
  set(CMAKE_HIP_ARCHITECTURES gfx1201 CACHE STRING "AMD GPU architecture")
  project(qtip_quant LANGUAGES CXX HIP)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(QTIP_CPU_ARCH "icelake-server" CACHE STRING "-march for the host fp64 Cholesky (AVX-512)")

find_package(OpenMP REQUIRED COMPONENTS CXX)
find_package(Threads REQUIRED)

# host-only core (codebook, hadamard reference, reference viterbi, packing, fp64 LDL, NVFP4 host codec)
add_library(qtip_core STATIC src/qtip_core.cpp src/log.cpp src/safetensors.cpp src/nvfp4.cpp)
target_include_directories(qtip_core PUBLIC src third_party)
target_compile_options(qtip_core PRIVATE -O3 -march=${QTIP_CPU_ARCH} -Wall -Wextra)
target_link_libraries(qtip_core PUBLIC OpenMP::OpenMP_CXX Threads::Threads)

if(QTIP_CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  find_package(CUDAToolkit REQUIRED)
  # the .hip sources are single-source: compile unmodified copies as CUDA (gpu.hpp switches on QTIP_CUDA)
  set(QTIP_GPU_KERNELS)
  foreach(fn IN ITEMS kernels attention viterbi)
    configure_file(${CMAKE_CURRENT_SOURCE_DIR}/src/${fn}.hip ${CMAKE_BINARY_DIR}/cuda/${fn}.cu COPYONLY)
    list(APPEND QTIP_GPU_KERNELS ${CMAKE_BINARY_DIR}/cuda/${fn}.cu)
  endforeach()
  add_library(qtip_gpu STATIC ${QTIP_GPU_KERNELS} src/gemm.cpp src/staging.cpp src/quantizer.cpp src/model.cpp
                               src/pipeline.cpp)
  target_compile_definitions(qtip_gpu PUBLIC QTIP_CUDA)
  target_include_directories(qtip_gpu PUBLIC src third_party)
  target_link_libraries(qtip_gpu PUBLIC qtip_core CUDA::cudart CUDA::cublasLt)
  target_compile_options(qtip_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>
                                          $<$<COMPILE_LANGUAGE:CXX>:-O3 -march=${QTIP_CPU_ARCH} -Wall>)
else()
  set(CMAKE_HIP_STANDARD 20)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  find_package(hip REQUIRED CONFIG)
  find_package(hipblaslt REQUIRED CONFIG)
  # GPU kernels + hipBLASLt wrapper + quantizer + model forward + pipeline
  add_library(qtip_gpu STATIC src/kernels.hip src/attention.hip src/viterbi.hip src/gemm.cpp src/staging.cpp
                              src/quantizer.cpp src/model.cpp src/pipeline.cpp)
  target_include_directories(qtip_gpu PUBLIC src third_party)
  target_link_libraries(qtip_gpu PUBLIC qtip_core hip::host roc::hipblaslt)
  target_compile_options(qtip_gpu PRIVATE $<$<COMPILE_LANGUAGE:HIP>:-mno-wavefrontsize64 -O3>
                                          $<$<COMPILE_LANGUAGE:CXX>:-O3 -march=${QTIP_CPU_ARCH} -Wall>)
endif()

add_executable(qtip_quant src/main.cpp)
target_link_libraries(qtip_quant PRIVATE qtip_gpu)
target_compile_options(qtip_quant PRIVATE -O3 -march=${QTIP_CPU_ARCH} -Wall)

add_executable(qtip_tests_cpu tests/test_cpu.cpp)
target_link_libraries(qtip_tests_cpu PRIVATE qtip_core)
add_executable(qtip_tests_gpu tests/test_gpu.cpp)
target_link_libraries(qtip_tests_gpu PRIVATE qtip_gpu)
add_executable(qtip_attention_tests tests/test_attention.cpp)
target_link_libraries(qtip_attention_tests PRIVATE qtip_gpu)
add_executable(qtip_adaptive_tests tests/test_adaptive.cpp)
target_link_libraries(qtip_adaptive_tests PRIVATE qtip_gpu)
if(QTIP_CUDA)
  add_executable(qtip_tests_cuda tests/test_cuda.cpp)
  target_link_libraries(qtip_tests_cuda PRIVATE qtip_gpu)
endif()

enable_testing()
add_test(NAME cpu COMMAND qtip_tests_cpu ${CMAKE_SOURCE_DIR}/tests/refvec)
add_test(NAME gpu COMMAND qtip_tests_gpu)
add_test(NAME gpu_attention COMMAND qtip_attention_tests)
add_test(NAME adaptive COMMAND qtip_adaptive_tests)
if(QTIP_CUDA)
  add_test(NAME cuda COMMAND qtip_tests_cuda)
endif()
set_tests_properties(gpu_attention PROPERTIES SKIP_RETURN_CODE 77)
